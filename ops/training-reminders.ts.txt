async function all(q: () => any) {
  let out: any[] = [], i = 0;
  for (;;) {
    const { data, error } = await q().range(i, i + 999);
    if (error) throw error;
    out = out.concat(data || []);
    if (!data || data.length < 1000) return out;
    i += 1000;
  }
}
// returns null when sent, otherwise the error; quota=true means stop for today
async function send(key: string, msg: Record<string, unknown>): Promise<{ err: string; quota: boolean } | null> {
  for (let n = 0; n < 3; n++) {

    if (r.ok) return null;
    const t = (await r.text()).slice(0, 300);
    if (r.status === 429 && /daily|quota|limit/i.test(t) && !/second/i.test(t)) return { err: "Resend daily limit: " + t, quota: true };
    if (r.status === 429) { await sleep(1500); continue; }
    return { err: "Resend " + r.status + ": " + t, quota: false };
  }
  return { err: "Resend rate limit", quota: false };
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: CORS });
  let body: any = {};
  if (req.method === "POST") { try { body = await req.json(); } catch (_) { body = {}; } }
  const mode = new URL(req.url).searchParams.get("mode") || body.mode || "send";
  if (!["send", "count", "sample", "cpr_request"].includes(mode)) return json({ error: "unknown mode" }, 400);
  const key = Deno.env.get("RESEND_API_KEY") || "";
  let admin = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") || "";
  if (!admin) { try { const k = JSON.parse(Deno.env.get("SUPABASE_SECRET_KEYS") || "{}"); admin = k.default || Object.values(k)[0] as string || ""; } catch (_) { /* none */ } }
  const sb = createClient(Deno.env.get("SUPABASE_URL")!, admin, { auth: { persistSession: false } });
  try {
    const { data: set } = await sb.from("reminder_settings").select("*").eq("id", 1).maybeSingle();
    const replyTo: string[] = set?.reply_to || ["torr@benevolenthouse.org", "hmashburn@benevolenthouse.org"];
    const staffTo: string[] = set?.staff_to || replyTo;
    const R = g.BH_REMIND;

    const tok = (req.headers.get("authorization") || "").replace(/^Bearer\s+/i, "");
    if (mode === "cpr_request") {
      // a signed-in caregiver asks to complete First Aid and CPR with Benevolent House
      if (!key) return json({ error: "email is not set up" }, 500);
      const { data: u } = await sb.auth.getUser(tok);
      const email = (u?.user?.email || "").toLowerCase();
      if (!email) return json({ error: "please sign in" }, 401);
      const { data: ap } = await sb.from("approved_people").select("email,full_name,home_label").eq("email", email).eq("active", true).maybeSingle();
      if (!ap) return json({ error: "not set up" }, 403);
      const { data: pr } = await sb.from("profiles").select("full_name,home_label").ilike("email", email).maybeSingle();
      const { data: recent } = await sb.from("reminder_log").select("id").eq("kind", "cpr").eq("email", email).gte("at", new Date(Date.now() - 864e5).toISOString()).limit(1);
      if (recent && recent.length) return json({ ok: true, already: true });
      const name = pr?.full_name || ap.full_name || email, home = pr?.home_label || ap.home_label || "";
      const e2 = (x: string) => String(x).replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" } as any)[c]);
      const r = await send(key, { from: FROM, to: staffTo, reply_to: [email], subject: "First Aid and CPR request from " + name + (home ? " (" + home + ")" : ""),
        html: `<div style="font-family:Arial;font-size:15px;line-height:1.5"><p><b>${e2(name)}</b>${home ? " (" + e2(home) + ")" : ""} would like to complete First Aid and CPR with Benevolent House.</p><p>Email: ${e2(email)}</p><p>Reply to this email to reach them directly. The class is $40 per caregiver with Benevolent House's HSI-certified trainers.</p></div>` });
      if (r) return json({ error: "could not send" }, 502);
      await sb.from("reminder_log").insert({ sent_on: new Date().toISOString().slice(0, 10), kind: "cpr", email, course_code: String(Date.now()), status: "sent" });
      return json({ ok: true });
    }
    if (mode === "sample") {
      // allowed when the caller's own token can read reminder_settings: a leadership account, or the Supabase dashboard
      let ok = false;
      if (tok) {
        try {
          const asCaller = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_ANON_KEY") || req.headers.get("apikey") || "", { global: { headers: { Authorization: "Bearer " + tok } }, auth: { persistSession: false } });
          const { data } = await asCaller.from("reminder_settings").select("id").eq("id", 1).maybeSingle();
          ok = !!data;
        } catch (_) { ok = false; }
      }
      if (!ok) return json({ error: "leadership only" }, 403);
    } else if (mode === "send") {
      const { data: recent } = await sb.from("reminder_log").select("id").eq("kind", "run").gte("at", new Date(Date.now() - 20 * 60e3).toISOString()).limit(1);
      if (recent && recent.length) return json({ skipped: "a run already happened in the last 20 minutes" });
      await sb.from("reminder_log").insert({ sent_on: new Date().toISOString().slice(0, 10), kind: "run", email: "", course_code: String(Date.now()), status: "started" });
    }

    const [people, profiles, comps, legacy] = await Promise.all([
      all(() => sb.from("approved_people").select("email,full_name,home_label,caregiver_type,access,active").order("email")),
      all(() => sb.from("profiles").select("id,email,full_name,home_label,caregiver_type").order("id")),
      all(() => sb.from("completions").select("id,user_id,course_code,completed_at,voided").order("id")),
      all(() => sb.from("legacy_training").select("id,person_name,course_code,completed_on,voided").order("id")),
    ]);
    const plan = R.plan({ people, profiles, comps, legacy });
    // marks already sent (or being sent) recently
    await sb.from("reminder_log").delete().eq("status", "sending").lt("at", new Date(Date.now() - 15 * 60e3).toISOString());
    const since = new Date(Date.now() - 10 * 864e5).toISOString().slice(0, 10);
    const logged = new Set((await all(() => sb.from("reminder_log").select("email,course_code,due_date,mark").eq("kind", "family").gte("sent_on", since).order("id"))).map((r: any) => r.email + "|" + r.course_code + "|" + r.due_date + "|" + r.mark));
    const open = (i: any) => i.marks.filter((m: number) => !logged.has(i.email + "|" + i.code + "|" + i.due + "|" + m));
    const fam = R.familyEmails(plan, (i: any) => open(i).length > 0);
    const staff = R.staffEmail(plan);

    if (mode === "count") return json({ today: plan.today, family_emails: fam.length, staff_items: staff ? staff.count : 0, enabled: !!set?.enabled, key_set: !!key });
    if (!key) return json({ error: "RESEND_API_KEY is not set" }, 500);
    if (mode === "sample") {
      const out: string[] = [];
      for (const e of fam.slice(0, 2)) { const r = await send(key, { from: FROM, to: staffTo, reply_to: replyTo, subject: "[Sample] " + e.subject, html: `<p style="font-family:Arial;color:#9A2F1F"><b>Sample only.</b> This is what a family receives. It was sent only to Benevolent House staff.</p>` + e.html }); out.push(r ? r.err : "sent"); await sleep(600); }
      if (staff) { const r = await send(key, { from: FROM, to: staffTo, reply_to: replyTo, subject: "[Sample] " + staff.subject, html: staff.html }); out.push(r ? r.err : "sent"); }
      return json({ today: plan.today, samples: out, family_emails_today: fam.length });
    }
    if (!set?.enabled) return json({ today: plan.today, skipped: "reminders are turned off" });

    let sent = 0, failed = 0, stopped = false;
    for (const e of fam) {
      const rows: any[] = [];
      e.items.forEach((i: any) => open(i).forEach((m: number) => rows.push({ sent_on: plan.today, kind: "family", email: e.to, course_code: i.code, due_date: i.due, mark: m })));
      const { data: claimed, error } = await sb.from("reminder_log").upsert(rows, { onConflict: "kind,email,course_code,due_date,mark", ignoreDuplicates: true }).select("id,course_code,due_date");
      if (error || !claimed || !claimed.length) continue;
      const got = new Set(claimed.map((c: any) => c.course_code + "|" + c.due_date));
      const mine = { today: plan.today, items: plan.items.filter((i: any) => i.email === e.to) };
      const msg = R.familyEmails(mine, (i: any) => got.has(i.code + "|" + i.due))[0];
      const ids = claimed.map((c: any) => c.id);
      if (!msg) { await sb.from("reminder_log").delete().in("id", ids); continue; }
      const r = await send(key, { from: FROM, to: [e.to], reply_to: replyTo, subject: msg.subject, html: msg.html });
      if (r) { await sb.from("reminder_log").delete().in("id", ids); failed++; if (r.quota) { stopped = true; break; } }
      else { await sb.from("reminder_log").update({ status: "sent" }).in("id", ids); sent++; }
      await sleep(600);
    }
    if (staff && !stopped) {
      const { data: claimed } = await sb.from("reminder_log").upsert([{ sent_on: plan.today, kind: "staff", email: "staff", course_code: "", due_date: plan.today, mark: -1 }], { onConflict: "kind,email,course_code,due_date,mark", ignoreDuplicates: true }).select("id");
      if (claimed && claimed.length) {
        const r = await send(key, { from: FROM, to: staffTo, reply_to: replyTo, subject: staff.subject, html: staff.html });
        if (r) await sb.from("reminder_log").delete().in("id", claimed.map((c: any) => c.id));
        else await sb.from("reminder_log").update({ status: "sent" }).in("id", claimed.map((c: any) => c.id));
      }
    }
    return json({ today: plan.today, family_sent: sent, family_failed: failed, stopped_for_daily_limit: stopped });
  } catch (e) {
    return json({ error: "reminder run failed" , detail: String((e as Error).message || e).slice(0, 200) }, 500);
  }
});
